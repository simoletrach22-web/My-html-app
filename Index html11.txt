<div id="ui">النقاط: 0 | الليفيل: 1 | الدم: 100</div>
<canvas id="game" width="400" height="600"></canvas>

<script>
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");
const ui = document.getElementById("ui");

// اللاعب الراجا
let player = { x: 200, y: 550, hp: 100 };

let zombies = [];
let bullets = [];
let score = 0;
let level = 1;
let zombieSpeed = 0.8;

// تحريك الراجا بالماوس
canvas.addEventListener("mousemove", (e) => {
const rect = canvas.getBoundingClientRect();
player.x = e.clientX - rect.left;
});

// الضرب
canvas.addEventListener("click", () => {
bullets.push({ x: player.x, y: player.y - 20 });
});

// كيخرج زومبي
function spawnZombie() {
zombies.push({
x: Math.random() * 360,
y: -40,
hp: level
});
}

// اللوب الرئيسي
function update() {
ctx.clearRect(0,0,400,600);

// رسم الراجا بالسلاح
ctx.font = "40px serif";
ctx.fillText("🧍🔫", player.x - 20, player.y);

// تحريك الزومبي و كيتبعو الراجا
zombies.forEach((z,i) => {
if(z.x < player.x) z.x += zombieSpeed;
if(z.x > player.x) z.x -= zombieSpeed;
z.y += zombieSpeed;

ctx.font = "35px serif";
ctx.fillText("🧟", z.x, z.y);

// إلا صد فالراجا
if(Math.abs(z.x - player.x) < 25 && Math.abs(z.y - player.y) < 25){
player.hp -= 10;
zombies.splice(i,1);
if(player.hp <= 0){
alert("متّي! النقاط: " + score);
document.location.reload();
}
}
});

// الطلقات
bullets.forEach((b,i) => {
b.y -= 8;
ctx.fillStyle = "yellow";
ctx.fillRect(b.x-2, b.y, 4, 10);

zombies.forEach((z,j) => {
if(Math.abs(b.x - z.x) < 20 && Math.abs(b.y - z.y) < 20){
z.hp--;
bullets.splice(i,1);
if(z.hp <= 0){
zombies.splice(j,1);
score++;
if(score % 10 === 0){
level++;
zombieSpeed += 0.3;
}
}
}
});
});

ui.innerText = النقاط: ${score} | الليفيل: ${level} | الدم: ${player.hp};
requestAnimationFrame(update);
}

setInterval(spawnZombie, 1200 - level*50);
update();
</script>
